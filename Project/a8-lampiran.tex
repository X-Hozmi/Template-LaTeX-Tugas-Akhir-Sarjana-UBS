\makeatletter
% Menggunakan 2 Parameter.
% Parameter #1 (Opsional/Dalam kurung siku): Untuk teks di Daftar Isi (Tanpa \\)
% Parameter #2 (Wajib/Dalam kurung kurawal): Untuk teks di Halaman Lampiran (Menggunakan \\ untuk piramida terbalik)
\renewcommand{\judulLampiran}[2][]{%
    \clearpage
    \refstepcounter{section}% Naikkan angka urutan lampiran (1, 2, 3...)

    % --- KODE KRUSIAL UNTUK MENGHITUNG TOTAL LAMPIRAN DI ABSTRAK ---
    \stepcounter{totallampiran}%
    % ----------------------------------------------------------------

    \phantomsection
    % Jika parameter opsional #1 diisi, gunakan itu untuk Daftar Isi. Jika kosong, gunakan #2.
    % \ifthenelse{\equal{#1}{}}{
    %     \addcontentsline{toc}{section}{\protect\numberline{}Lampiran \arabic{section}. #2}%
    % }{
    %     \addcontentsline{toc}{section}{\protect\numberline{}Lampiran \arabic{section}. #1}%
    % }

    % Mencetak teks di halaman dokumen (Rata Tengah, Kapital, 1 Spasi)
    \begin{center}
        \textbf{\large LAMPIRAN \arabic{section}} \\[1\baselineskip]
        \textbf{\large \MakeUppercase{#2}}
    \end{center}
    \vspace{1.5em}
}
\makeatother

% Mereset nomor menjadi 1, 2, 3
\setcounter{section}{0}
\renewcommand{\thesection}{\arabic{section}}
